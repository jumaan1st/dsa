<p>import java.util.HashMap;</p><p>import <a target="_blank" rel="noopener noreferrer nofollow" href="http://java.util.Map">java.util.Map</a>;</p><p>class Solution {</p><p>    public int[] twoSum(int[] nums, int target) {</p><p>        Map&lt;Integer, Integer&gt; map = new HashMap&lt;&gt;();</p><p>        for (int i = 0; i &lt; nums.length; i++) {</p><p>            int complement = target - nums[i];</p><p>            if (map.containsKey(complement)) {</p><p>                return new int[]{ map.get(complement), i };</p><p>            }</p><p>            map.put(nums[i], i);</p><p>        }</p><p>        return new int[]{}; // never reached as per problem statement</p><p>    }</p><p>}</p><p></p>
